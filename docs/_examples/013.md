[#*phrase*\
  HEAD\t|1|\
  SPR\t〈<>〉\
  COMPS\t〈<>〉
  [#*word*\
    HEAD\t|2|<>*noun*\
    SPR\t〈<>〉\
    COMPS\t〈<>〉
    Kim
  ]
  [#*phrase*\
    HEAD\t|1|\
    SPR\t〈<>|2|<>〉\
    COMPS\t〈<>〉
    [#*word*\
      HEAD\t|1|<>*verb*\
      SPR\t〈<>|2|<>NP〉\
      COMPS\t〈<>|3|<>PP\[*on*\]〉
      relies
    ]
    [#|3|<>*phrase*\
      HEAD\t|4|\
      SPR\t〈<>〉\
      COMPS\t〈<>〉
      [#*word*\
        HEAD\t|4|<>*prep*\
        FORM\t*on*\
        SPR\t〈<>〉\
        COMPS\t〈<>|5|<>〉
        on
      ]
      [#|5|<>*word*\
        HEAD\t*noun*\
        SPR\t〈<>〉\
        COMPS\t〈<>〉
        Sandy
      ]
    ]
  ]
]
